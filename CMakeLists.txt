cmake_minimum_required(VERSION 3.16)
project(margin_call LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(MC_SANITIZE "Build with address and undefined-behaviour sanitizers" OFF)
option(MC_WERROR "Treat warnings as errors" OFF)

add_library(mc INTERFACE)
target_include_directories(mc INTERFACE include)
target_compile_options(mc INTERFACE -Wall -Wextra -Wpedantic -Wshadow -Wconversion)
if(MC_WERROR)
  target_compile_options(mc INTERFACE -Werror)
endif()
if(MC_SANITIZE)
  target_compile_options(mc INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
  target_link_options(mc INTERFACE -fsanitize=address,undefined)
endif()

foreach(app backtest gensynth bench server)
  add_executable(mc_${app} apps/${app}.cpp)
  target_link_libraries(mc_${app} PRIVATE mc)
endforeach()
set_target_properties(mc_backtest PROPERTIES OUTPUT_NAME mc_backtest)

add_executable(mc_tests tests/test_main.cpp)
target_link_libraries(mc_tests PRIVATE mc)
enable_testing()
add_test(NAME mc_tests COMMAND mc_tests)
